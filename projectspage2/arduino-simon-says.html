03

How the game works

The Arduino turns on LEDs in a pattern. The player watches the pattern, then presses the matching buttons in the same order. Each round adds one more step to the sequence. If the player presses the wrong button or takes too long, the game ends.

#define CHOICE_OFF      0

#define CHOICE_BLUE     (1 << 2)
#define CHOICE_YELLOW   (1 << 3)

#define LED_BLUE    13
#define LED_YELLOW  5

#define BUTTON_RED    9
#define BUTTON_GREEN  2
#define BUTTON_BLUE   12
#define BUTTON_YELLOW 6

#define BUZZER1  4
#define BUZZER2  7

#define ROUNDS_TO_WIN      13
#define ENTRY_TIME_LIMIT   3000

byte gameBoard[32];
byte gameRound = 0;

void setup() {
    Serial.begin(9600);

    pinMode(BUTTON_RED, INPUT_PULLUP);
    pinMode(BUTTON_GREEN, INPUT_PULLUP);
    pinMode(BUTTON_BLUE, INPUT_PULLUP);
    pinMode(BUTTON_YELLOW, INPUT_PULLUP);

    pinMode(LED_RED, OUTPUT);
    pinMode(LED_GREEN, OUTPUT);
    pinMode(LED_BLUE, OUTPUT);
    pinMode(LED_YELLOW, OUTPUT);

    pinMode(BUZZER1, OUTPUT);
    pinMode(BUZZER2, OUTPUT);
}
